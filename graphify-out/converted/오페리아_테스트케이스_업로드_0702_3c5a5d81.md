## Sheet: testcases
| 질문 | Gold SQL |
| --- | --- |
| 현재 사용할 수 있는 금액 알려줘 | SELECT ACCOUNT_BALANCE_AMT AS available_amount
FROM V_USER_ASSET
WHERE USER_ID = 'USR131' |
| 월별 사용 내역 알려줘 | SELECT DLNG_YM AS month, SUM(DLNG_AMT) AS sum_pay_dlng_amt
FROM V_PAY_USE
WHERE USER_ID = 'USR131'
GROUP BY DLNG_YM
ORDER BY month ASC |
| 근처 사용 가능 가맹점 알려줘 | SELECT
  MERCHANT_NM AS merchant_nm,
  ADDR        AS addr,
  CATEGORY_CD AS category_cd,
  LATITUDE    AS latitude,
  LONGITUDE   AS longitude,
  ROUND(6371000 * ACOS(
    COS(RADIANS(36.6424)) * COS(RADIANS(LATITUDE)) *
    COS(RADIANS(LONGITUDE) - RADIANS(127.4890)) +
    SIN(RADIANS(36.6424)) * SIN(RADIANS(LATITUDE))
  )) AS distance_m
FROM V_MERCHANT
WHERE USE_YN = 'Y' AND VOUCHER_STTS_NM = '정상' AND LATITUDE IS NOT NULL
ORDER BY distance_m ASC
LIMIT 20 |
| 사용처별 내역 알려줘 | SELECT MERCHANT_NM AS merchant_nm, SUM(DLNG_AMT) AS sum_pay_dlng_amt
FROM V_PAY_USE
WHERE USER_ID = 'USR131'
GROUP BY MERCHANT_NM
ORDER BY sum_pay_dlng_amt DESC |
